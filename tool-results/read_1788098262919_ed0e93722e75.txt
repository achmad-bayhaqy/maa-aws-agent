   743→   743→def handle_confirm(payload):
   744→   744→    sid = payload["sessionId"]
   745→   745→    user_id = payload["userId"]
   746→   746→    token = payload["confirmToken"]
   747→   747→    t1 = payload.get("typed1", "")
   748→   748→    t2 = payload.get("typed2", "")
   749→   749→
   750→   750→    r = get_client("dynamodb").get_item(TableName=CONF_TABLE, Key={"confirmToken": {"S": token}})
   751→   751→    rec = r.get("Item")
   752→   752→    if not rec:
   753→   753→        return {"sessionId": sid, "status": "error", "message": "Token konfirmasi tidak ditemukan / kedaluwarsa."}
   754→   754→    if rec["status"]["S"] != "pending":
   755→   755→        return {"sessionId": sid, "status": "error", "message": f"Status konfirmasi: {rec['status']['S']}."}
   756→   756→    if rec["userId"]["S"] != user_id or rec["sessionId"]["S"] != sid:
   757→   757→        return {"sessionId": sid, "status": "error", "message": "Konfirmasi bukan milik sesi Anda."}
   758→   758→    exp = int(rec["expiresAt"]["N"]) * 1000
   759→   759→    if now_ms() > exp:
   760→   760→        get_client("dynamodb").update_item(TableName=CONF_TABLE, Key={"confirmToken": {"S": token}},
   761→   761→                        UpdateExpression="SET #s = :v", ExpressionAttributeNames={"#s": "status"},
   762→   762→                        ExpressionAttributeValues={":v": {"S": "expired"}})
   763→   763→        return {"sessionId": sid, "status": "error", "message": "Jendela konfirmasi 5 menit lewat."}
   764→   764→    chal = rec["challenge"]["S"]
   765→   765→    if t1 != chal or t2 != chal:
   766→   766→        put_trace(sid, "error", "Konfirmasi gagal: string tidak cocok")
   767→   767→        return {"sessionId": sid, "status": "mismatch",
   768→   768→                "message": "String konfirmasi tidak cocok. Ketik persis dua kali."}
   769→   769→
   770→   770→    op = json.loads(rec["operation"]["S"])
   771→   771→    tname, targs = op["tool"], op["input"]
   772→   772→    put_trace(sid, "thinking", f"Konfirmasi ganda OK - mengeksekusi {tname} {targs}")
   773→   773→    try:
   774→   774→        result = exec_tool(tname, targs)
   775→   775→    except Exception as e:
   776→   776→        result = {"status": "error", "message": str(e)[:300]}
   777→   777→    get_client("dynamodb").update_item(TableName=CONF_TABLE, Key={"confirmToken": {"S": token}},
   778→   778→                    UpdateExpression="SET #s = :v, #res = :r",
   779→   779→                    ExpressionAttributeNames={"#s": "status", "#res": "result"},
   780→   780→                    ExpressionAttributeValues={":v": {"S": "executed" if result.get("status") == "ok" else "failed"},
   781→   781→                                               ":r": {"S": json.dumps(result, ensure_ascii=False)[:3000]}})
   782→   782→    put_trace(sid, "confirm_executed", f"{tname} {targs} -> {json.dumps(result, ensure_ascii=False)[:600]}")
   783→   783→
   784→   784→    rec2 = session_get(sid)
   785→   785→    prev = []
   786→   786→    if rec2 and "messages" in rec2:
   787→   787→        for m in rec2["messages"]["L"]:
   788→   788→            prev.append({"role": m["M"]["role"]["S"], "text": m["M"]["text"]["S"], "ts": int(m["M"]["ts"]["N"])})
   789→   789→    ok = result.get("status") == "ok"
   790→   790→    sysmsg = ("✅ Eksekusi berhasil: " if ok else "❌ Eksekusi gagal: ") + tname + " " + json.dumps(targs, ensure_ascii=False)
   791→   791→    if ok and tname == "aws_delete_resource":
   792→   792→        sysmsg += f". Resource {targs.get('identifier')} telah dihapus permanen."
   793→   793→    messages_db = prev + [{"role": "user", "text": f"[SISTEM] {sysmsg}"}]
   794→   794→    session_put(sid, user_id, payload.get("username", "user"), "done", messages_db,
   795→   795→                extra={"createdAt": rec2["createdAt"]["S"] if rec2 else str(now_ms())})
   796→   796→    return {"sessionId": sid, "status": "executed", "result": result}

   808→   808→class Handler(BaseHTTPRequestHandler):
   809→   809→    def _send(self, code, obj):
   810→   810→        body = json.dumps(obj, ensure_ascii=False, default=str).encode()
   811→   811→        self.send_response(code)
   812→   812→        self.send_header("Content-Type", "application/json")
   813→   813→        self.send_header("Content-Length", str(len(body)))
   814→   814→        self.end_headers()
   815→   815→        self.wfile.write(body)

   818→   818→        self._send(200, {"status": "Healthy"})

   821→   821→        n = int(self.headers.get("Content-Length", 0))
   822→   822→        raw = self.rfile.read(n)
   823→   823→        try:
   824→   824→            payload = json.loads(raw or b"{}")
   825→   825→        except Exception:
   826→   826→            payload = {}
   827→   827→        try:
   828→   828→            result = invoke(payload, None)
   829→   829→            self._send(200, result)
   830→   830→        except Exception as e:
   831→   831→            self._send(500, {"error": str(e)[:500]})
