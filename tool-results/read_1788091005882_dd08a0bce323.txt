    54→    54→      } else {
    55→    55→        setAuthSession(r.session);
    56→    56→        setView('mfa_challenge');

    60→    60→    } finally {
    61→    61→      setBusy(false);
    62→    62→    }

    70→    70→      saveSession(username.trim(), t);
    71→    71→      setTokens(t);
    72→    72→      setInfo('MFA TOTP berhasil didaftarkan!');
    73→    73→      setView('chat');
    74→    74→    } catch (e) {
    75→    75→      setError((e as Error).message);
    76→    76→    } finally {
    77→    77→      setBusy(false);

    86→    86→      setTokens(t);
    87→    87→      setView('chat');
    88→    88→    } catch (e) {
    89→    89→      setError((e as Error).message);
    90→    90→    } finally {
    91→    91→      setBusy(false);
    92→    92→    }
    93→    93→  };
    94→    94→
    95→    95→  if (view === 'login')
    96→    96→    return (
    97→    97→      <Shell>
    98→    98→        <LoginCard username={username} setUsername={setUsername} password={password} setPassword={setPassword}
    99→    99→          onSubmit={doLogin} busy={busy} error={error} info={info} />
   100→   100→      </Shell>

   102→   102→

   104→   104→    return (
   105→   105→      <Shell>
   106→   106→        <MfaSetupCard qrData={qrData} secret={secret} code={code} setCode={setCode}
   107→   107→          onVerify={doVerifyEnroll} busy={busy} error={error} />
   108→   108→      </Shell>

   110→   110→

   114→   114→        <MfaChallengeCard code={code} setCode={setCode} onSubmit={doMfaChallenge} busy={busy} error={error}

   218→   218→
   219→   219→function ChatApp({ tokens, username, onLogout, setErrorTop }: {
   220→   220→  tokens: Tokens; username: string; onLogout: (msg?: string) => void; setErrorTop: (s: string) => void;
   221→   221→}) {
   222→   222→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   223→   223→  const [input, setInput] = useState('');
   224→   224→  const [mode, setMode] = useState<Mode>('FAST');
   225→   225→  const [manualModel, setManualModel] = useState('');
   226→   226→  const [models, setModels] = useState<{ modelId: string; provider: string; name: string }[]>([]);
   227→   227→  const [sessionId, setSessionId] = useState<string | null>(null);
   228→   228→  const [processing, setProcessing] = useState(false);
   229→   229→  const [traceEvents, setTraceEvents] = useState<TraceEvent[]>([]);
   230→   230→  const [showTrace, setShowTrace] = useState(false);
   231→   231→  const [pendingConfirm, setPendingConfirm] = useState<PendingConfirm | null>(null);
   232→   232→  const [typed1, setTyped1] = useState('');
   233→   233→  const [typed2, setTyped2] = useState('');
   234→   234→  const [confirmBusy, setConfirmBusy] = useState(false);
   235→   235→  const [confirmMsg, setConfirmMsg] = useState('');
   236→   236→  const [showKb, setShowKb] = useState(false);
   237→   237→  const [showMenu, setShowMenu] = useState(false);
   238→   238→  const [sessions, setSessions] = useState<{ sessionId: string; title: string; status: string; mode: string }[]>([]);
   239→   239→  const [kbDocs, setKbDocs] = useState<{ key: string; name: string; size: number }[]>([]);
   240→   240→  const [kbBusy, setKbBusy] = useState('');
   241→   241→  const pollRef = useRef<{ stop: boolean } | null>(null);
   242→   242→  const scrollRef = useRef<HTMLDivElement>(null);
   243→   243→  const lastActivity = useRef(Date.now());
   244→   244→
   245→   245→  const finishSession = useCallback(async (msg?: string) => {
   246→   246→    if (pollRef.current) pollRef.current.stop = true;
   247→   247→    const s = loadSession();
   248→   248→    if (s?.tokens?.RefreshToken) await revokeToken(s.tokens.RefreshToken);
   249→   249→    try { await signOutAll(tokens.IdToken); } catch { /* best effort */ }
   250→   250→    onLogout(msg);
   251→   251→  }, [tokens?.IdToken, onLogout]);
   252→   252→
   253→   253→  // ------- idle 15 menit (FR 1.4) -------
   254→   254→  useEffect(() => {
   255→   255→    const bump = () => { lastActivity.current = Date.now(); };
   256→   256→    const evs = ['keydown', 'touchstart', 'click', 'scroll'];
   257→   257→    evs.forEach((e) => window.addEventListener(e, bump, { passive: true }));
   258→   258→    const iv = setInterval(() => {
   259→   259→      if (Date.now() - lastActivity.current > CONFIG.idleTimeoutMs) {
   260→   260→        finishSession('Sesi hangus otomatis setelah 15 menit tidak ada aktivitas.');
   261→   261→      }
   262→   262→    }, 10_000);
   263→   263→    return () => { evs.forEach((e) => window.removeEventListener(e, bump)); clearInterval(iv); };
   264→   264→  }, [finishSession]);
   265→   265→
   266→   266→  // load models
   267→   267→  useEffect(() => {
   268→   268→    getModels(tokens.IdToken).then((r) => {
   269→   269→      setModels(r.models);
   270→   270→      if (r.models.length) setManualModel('zai.glm-5');
   271→   271→    }).catch(() => {});
   272→   272→  }, [tokens.IdToken]);
   273→   273→
   274→   274→  // open previous session
   275→   275→  const openSession = useCallback(async (sid: string) => {
   276→   276→    try {
   277→   277→      const s = await getStatus(tokens.IdToken, sid);
   278→   278→      setSessionId(sid);
   279→   279→      setMessages(s.messages || []);
   280→   280→      setTraceEvents([]);
   281→   281→      setPendingConfirm(s.pendingConfirmation || null);
   282→   282→      if (s.mode === 'DEEP' || s.mode === 'MANUAL') setMode(s.mode as Mode);
   283→   283→    } catch (e) { setErrorTop((e as Error).message); }
   284→   284→  }, [tokens.IdToken]);
   285→   285→
   286→   286→  // ------- polling -------
   287→   287→  const startPolling = useCallback((sid: string) => {
   288→   288→    const handle = { stop: false };
   289→   289→    pollRef.current = handle;
   290→   290→    let cursor = 0;
   291→   291→    const tick = async () => {
   292→   292→      if (handle.stop) return;
   293→   293→      try {
   294→   294→        const tr = await getTrace(tokens.IdToken, sid, cursor);
   295→   295→        if (tr.events?.length) {
   296→   296→          setTraceEvents((prev) => [...prev, ...tr.events]);
   297→   297→          cursor = Number(tr.events[tr.events.length - 1].ts) + 1;
   298→   298→        }
   299→   299→        const st = await getStatus(tokens.IdToken, sid);
   300→   300→        if (st.pendingConfirmation) setPendingConfirm(st.pendingConfirmation);
   301→   301→        if (st.status === 'done' || st.status === 'error') {
   302→   302→          setMessages(st.messages || []);
   303→   303→          setProcessing(false);
   304→   304→          setPendingConfirm(st.pendingConfirmation || null);
   305→   305→          handle.stop = true;
   306→   306→          return;
   307→   307→        }
   308→   308→      } catch { /* jaringan flaky: lanjut polling */ }
   309→   309→      setTimeout(tick, 2000);
   310→   310→    };
   311→   311→    setTimeout(tick, 900);
   312→   312→  }, [tokens.IdToken]);
   313→   313→
   314→   314→  const send = async () => {
   315→   315→    const msg = input.trim();
   316→   316→    if (!msg || processing) return;
   317→   317→    setInput('');
   318→   318→    setMessages((m) => [...m, { role: 'user', text: msg, ts: Date.now() }]);
   319→   319→    setTraceEvents([]);
   320→   320→    setPendingConfirm(null);
   321→   321→    setProcessing(true);
   322→   322→    setShowTrace(true);
   323→   323→    try {
   324→   324→      const r = await sendChat(tokens.IdToken, msg, mode, mode === 'MANUAL' ? manualModel : undefined);
   325→   325→      setSessionId(r.sessionId);
   326→   326→      startPolling(r.sessionId);
   327→   327→    } catch (e) {
   328→   328→      setProcessing(false);
   329→   329→      setErrorTop((e as Error).message);
   330→   330→    }
   331→   331→  };
   332→   332→
   333→   333→  const doConfirm = async () => {
   334→   334→    if (!pendingConfirm || !sessionId) return;
   335→   335→    setConfirmBusy(true); setConfirmMsg('');
   336→   336→    try {
   337→   337→      const r = await confirmDestructive(tokens.IdToken, sessionId, pendingConfirm.confirmToken, typed1, typed2);
   338→   338→      if (r.status === 'executed') {
   339→   339→        setPendingConfirm(null); setTyped1(''); setTyped2('');
   340→   340→        const st = await getStatus(tokens.IdToken, sessionId);
   341→   341→        setMessages(st.messages || []);
   342→   342→        setShowTrace(true);
   343→   343→      } else {
   344→   344→        setConfirmMsg(r.message || 'Konfirmasi gagal');
   345→   345→      }
   346→   346→    } catch (e) {
   347→   347→      setConfirmMsg((e as Error).message);
   348→   348→    } finally {
   349→   349→      setConfirmBusy(false);
   350→   350→    }
   351→   351→  };
   352→   352→
   353→   353→  // ------- KB -------
   354→   354→  const refreshKb = useCallback(async () => {
   355→   355→    try {
   356→   356→      const r = await listKbDocs(tokens.IdToken);
   357→   357→      setKbDocs(r.docs);
   358→   358→    } catch { /* ignore */ }
   359→   359→  }, [tokens.IdToken]);
   360→   360→
   361→   361→  const uploadDoc = async (file: File) => {
   362→   362→    setKbBusy('mengunggah…');
   363→   363→    try {
   364→   364→      const { uploadUrl, key } = await presignUpload(tokens.IdToken, file.name, file.type || 'application/octet-stream');
   365→   365→      await fetch(uploadUrl, { method: 'PUT', body: file, headers: { 'Content-Type': file.type || 'application/octet-stream' } });
   366→   366→      setKbBusy('sinkronisasi KB…');
   367→   367→      await syncKb(tokens.IdToken);
   368→   368→      setKbBusy('');
   369→   369→      await refreshKb();
   370→   370→      setInfo2(`Dokumen ${file.name} terunggah & KB sedang disinkronkan`);
   371→   371→    } catch (e) {
   372→   372→      setKbBusy('');
   373→   373→      setInfo2(`Upload gagal: ${(e as Error).message}`);
   374→   374→    }
   375→   375→  };
   376→   376→
   377→   377→  const [info2, setInfo2] = useState('');
   378→   378→  useEffect(() => { if (info2) { const t = setTimeout(() => setInfo2(''), 4000); return () => clearTimeout(t); } }, [info2]);
   379→   379→
   380→   380→  const openMenu = async () => {
   381→   381→    setShowMenu(true);
   382→   382→    try {
   383→   383→      const r = await getSessions(tokens.IdToken);
   384→   384→      setSessions(r.sessions);
   385→   385→    } catch { /* ignore */ }
   386→   386→  };
   387→   387→
   388→   388→  useEffect(() => { scrollRef.current?.scrollTo({ top: 999999, behavior: 'smooth' }); }, [messages.length]);
   389→   389→
   390→   390→  const groupedModels = useMemo(() => {
   391→   391→    const g: Record<string, typeof models> = {};
   392→   392→    models.forEach((m) => { (g[m.provider] = g[m.provider] || []).push(m); });
   393→   393→    return g;
   394→   394→  }, [models]);
