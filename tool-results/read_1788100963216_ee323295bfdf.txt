    35→    35→/** Normalisasi status: ekstrak blok [[CLARIFY]] dari pesan terakhir. */
    36→    36→function normalizeStatus(st: ChatStatus): ChatStatus {
    37→    37→  const msgs: ChatMessage[] = [...(st.messages || [])];
    38→    38→  let clarify = st.clarify ?? null;
    39→    39→  if (!clarify && msgs.length) {
    40→    40→    for (let i = msgs.length - 1; i >= 0; i--) {
    41→    41→      if (msgs[i].role === 'assistant') {
    42→    42→        const { text, clarify: c } = stripClarifyBlock(msgs[i].text);
    43→    43→        if (c) {
    44→    44→          msgs[i] = { ...msgs[i], text };
    45→    45→          clarify = c;
    46→    46→        }
    47→    47→        break;
    48→    48→      }
    49→    49→    }
    50→    50→  }
    51→    51→  return { ...st, messages: msgs, clarify };
    52→    52→}
    53→    53→
    54→    54→function traceLastTs(events: TraceEvent[]): number {
    55→    55→  let last = 0;
    56→    56→  for (const e of events) {
    57→    57→    const n = Number(e.ts);
    58→    58→    if (Number.isFinite(n) && n > last) last = n;
    59→    59→  }
    60→    60→  return last;
    61→    61→}
